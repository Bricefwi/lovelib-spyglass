/**
 * Composant ProtectedRoute
 * 
 * Protège les routes sensibles avec vérifications multiples :
 * - Authentification
 * - Confirmation email
 * - Onboarding complété (SIRET validé)
 * - Statut du compte
 */

import { ReactNode, useEffect, useState } from "react";
import { useNavigate } from "react-router-dom";
import { useUserRole } from "@/hooks/useUserRole";
import { supabase } from "@/integrations/supabase/client";
import { Loader2 } from "lucide-react";
import { toast } from "sonner";

interface ProtectedRouteProps {
  children: ReactNode;
  requireAuth?: boolean;
  requireAdmin?: boolean;
  requireClient?: boolean;
  requireOnboarding?: boolean;
  redirectTo?: string;
}

export function ProtectedRoute({
  children,
  requireAuth = false,
  requireAdmin = false,
  requireClient = false,
  requireOnboarding = false,
  redirectTo = "/auth",
}: ProtectedRouteProps) {
  const navigate = useNavigate();
  const { role, loading: roleLoading, isAdmin, isClient } = useUserRole();
  const [isChecking, setIsChecking] = useState(true);
  const [isAuthorized, setIsAuthorized] = useState(false);

  useEffect(() => {
    const checkAccess = async () => {
      try {
        // 1. Vérifier l'authentification
        const { data: { session }, error: sessionError } = await supabase.auth.getSession();
        
        if (sessionError || !session) {
          if (requireAuth || requireAdmin || requireClient || requireOnboarding) {
            toast.error("Veuillez vous connecter pour accéder à cette page");
            navigate(redirectTo);
            return;
          }
        }

        const user = session?.user;

        // 2. Vérifier la confirmation email
        if (user && requireOnboarding && !user.email_confirmed_at) {
          toast.error("Veuillez confirmer votre email avant de continuer. Vérifiez votre boîte mail.");
          navigate('/auth');
          return;
        }

        // 3. Vérifier l'onboarding (si requis)
        if (user && requireOnboarding) {
          const { data: profile, error: profileError } = await supabase
            .from('profiles')
            .select('company_siret, account_status, blocked_until, trust_score')
            .eq('user_id', user.id)
            .single();

          if (profileError) {
            console.error('Error fetching profile:', profileError);
            toast.error("Impossible de récupérer votre profil");
            navigate('/auth');
            return;
          }

          // 4. Vérifier que l'onboarding est complété
          if (!profile?.company_siret) {
            toast.info("Veuillez compléter votre profil entreprise pour continuer");
            navigate('/onboarding/company');
            return;
          }

          // 5. Vérifier que le compte n'est pas bloqué
          if (profile.blocked_until && new Date(profile.blocked_until) > new Date()) {
            const blockedUntilDate = new Date(profile.blocked_until).toLocaleString('fr-FR');
            toast.error(`Compte temporairement bloqué jusqu'au ${blockedUntilDate}`);
            navigate('/');
            return;
          }

          // 6. Vérifier le statut du compte
          if (profile.account_status === 'rejected') {
            toast.error("Votre demande d'accès a été rejetée. Contactez le support.");
            navigate('/');
            return;
          }

          // 7. Avertir si trust score faible (mais ne pas bloquer)
          if (profile.trust_score < 30) {
            toast.warning("Votre compte est en cours de vérification. Certaines fonctionnalités peuvent être limitées.");
          }
        }

        // Attendre que le rôle soit chargé avant de vérifier les permissions admin/client
        if (roleLoading) return;

        // Vérifier le rôle admin si requis
        if (requireAdmin && !isAdmin) {
          toast.error("Accès non autorisé - Réservé aux administrateurs");
          navigate("/");
          return;
        }

        // Vérifier le rôle client si requis
        if (requireClient && !isClient && !isAdmin) {
          toast.error("Accès non autorisé - Réservé aux clients");
          navigate("/");
          return;
        }

        // Toutes les vérifications passées
        setIsAuthorized(true);
      } catch (error) {
        console.error('Error in ProtectedRoute:', error);
        toast.error("Une erreur est survenue lors de la vérification de vos accès");
        navigate('/');
      } finally {
        setIsChecking(false);
      }
    };

    checkAccess();
  }, [roleLoading, role, isAdmin, isClient, requireAuth, requireAdmin, requireClient, requireOnboarding, navigate, redirectTo]);

  // Afficher un loader pendant la vérification
  if (isChecking || roleLoading) {
    return (
      <div className="min-h-screen flex items-center justify-center">
        <div className="text-center">
          <Loader2 className="h-8 w-8 animate-spin mx-auto mb-4 text-primary" />
          <p className="text-sm text-muted-foreground">Vérification de vos accès...</p>
        </div>
      </div>
    );
  }

  // Afficher le contenu si autorisé
  return isAuthorized ? <>{children}</> : null;
}

/**
 * HOC pour protéger une route admin
 */
export function withAdminProtection<P extends object>(
  Component: React.ComponentType<P>
) {
  return function ProtectedComponent(props: P) {
    return (
      <ProtectedRoute requireAuth requireAdmin>
        <Component {...props} />
      </ProtectedRoute>
    );
  };
}

/**
 * HOC pour protéger une route client
 */
export function withClientProtection<P extends object>(
  Component: React.ComponentType<P>
) {
  return function ProtectedComponent(props: P) {
    return (
      <ProtectedRoute requireAuth requireClient>
        <Component {...props} />
      </ProtectedRoute>
    );
  };
}

/**
 * HOC pour protéger une route authentifiée (sans rôle spécifique)
 */
export function withAuthProtection<P extends object>(
  Component: React.ComponentType<P>
) {
  return function ProtectedComponent(props: P) {
    return (
      <ProtectedRoute requireAuth>
        <Component {...props} />
      </ProtectedRoute>
    );
  };
}

# src/hooks/useUserRole.tsx
# ------------------

import { useState, useEffect } from "react";
import { supabase } from "@/integrations/supabase/client";
import { Database } from "@/integrations/supabase/types";

type AppRole = Database["public"]["Enums"]["app_role"];

export const useUserRole = () => {
  const [role, setRole] = useState<AppRole | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const fetchRole = async () => {
      try {
        const { data: { session } } = await supabase.auth.getSession();
        
        if (!session?.user) {
          setRole(null);
          setLoading(false);
          return;
        }

        const { data, error } = await supabase
          .from("user_roles")
          .select("role")
          .eq("user_id", session.user.id)
          .single();

        if (error) {
          console.error("Error fetching role:", error);
          setRole(null);
        } else {
          setRole(data?.role || null);
        }
      } catch (error) {
        console.error("Error:", error);
        setRole(null);
      } finally {
        setLoading(false);
      }
    };

    fetchRole();

    const { data: { subscription } } = supabase.auth.onAuthStateChange(() => {
      fetchRole();
    });

    return () => subscription.unsubscribe();
  }, []);

  return { role, loading, isAdmin: role === "admin", isClient: role === "client" };
};

const Dashboard = () => {
  const [profile, setProfile] = useState<ProfileData | null>(null);
  const [loading, setLoading] = useState(true);
  const [audits, setAudits] = useState<AuditData[]>([]);
  const [stats, setStats] = useState({
    total: 0,
    completed: 0,
    inProgress: 0,
    pending: 0
  });
  const navigate = useNavigate();
  const { isAdmin } = useUserRole();

  // Redirection automatique des admins vers leur tableau de bord dédié
  useEffect(() => {
    if (isAdmin) {
      navigate("/admin/dashboard", { replace: true });
    }
  }, [isAdmin, navigate]);

  useEffect(() => {
    fetchDashboardData();
  }, []);

  const fetchDashboardData = async () => {
    try {
      const { data: { user } } = await supabase.auth.getUser();
      
      if (!user) {
        navigate('/auth');
        return;
      }

      // ✅ Récupérer le profil de l'utilisateur
      const { data: profileData, error: profileError } = await supabase
        .from('profiles')
        .select('*')
        .eq('user_id', user.id)
        .single();

      if (profileError) throw profileError;
      setProfile(profileData);

      // ✅ Récupérer UNIQUEMENT les audits de cet utilisateur
      const { data: auditsData, error: auditsError } = await supabase
        .from('pre_audits')
        .select(`
          id,
          url,
          status,
          confidence,
          created_at,
          company:companies(name, website)
        `)
        .eq('created_by', user.id)
        .order('created_at', { ascending: false })
        .limit(5);

      if (auditsError) throw auditsError;
      setAudits(auditsData || []);

      // ✅ Calculer les statistiques personnelles
      const { data: allAudits } = await supabase
        .from('pre_audits')
        .select('status')
        .eq('created_by', user.id);

      const total = allAudits?.length || 0;
      const completed = allAudits?.filter(a => a.status === 'consolidated').length || 0;
      const inProgress = allAudits?.filter(a => 
        ['collecting', 'ai_running', 'consolidating'].includes(a.status)
      ).length || 0;
      const pending = allAudits?.filter(a => a.status === 'new' || a.status === 'pending').length || 0;

      setStats({ total, completed, inProgress, pending });

    } catch (error) {
      console.error('Error fetching dashboard data:', error);
      toast.error('Erreur lors du chargement du tableau de bord');
    } finally {
      setLoading(false);
    }
  };

  const viewReport = async (preAuditId: string) => {
    try {
      // Vérifier que l'utilisateur possède cet audit
      const { data: { user } } = await supabase.auth.getUser();
      if (!user) return;

      const { data: auditCheck } = await supabase
        .from("pre_audits")
        .select("created_by")
        .eq("id", preAuditId)
        .single();

      if (!auditCheck || auditCheck.created_by !== user.id) {
        toast.error("Accès non autorisé");
        return;
      }

      const { data, error } = await supabase
        .from("audit_reports")
        .select("id")
        .eq("pre_audit_id", preAuditId)
        .order("created_at", { ascending: false })
        .limit(1)
        .single();

      if (error || !data) {
        toast.error("Aucun rapport disponible");
        return;
      }

      window.open(`/audit-report/${data.id}`, "_blank");
    } catch (error) {
      console.error("Error viewing report:", error);
      toast.error("Erreur lors de l'ouverture du rapport");
    }
  };

  const getTrustScoreColor = (score: number) => {
    if (score >= 80) return 'text-green-600';
    if (score >= 50) return 'text-yellow-600';
    return 'text-red-600';
  };

  const getStatusBadge = (status: string) => {
    const statusConfig: Record<string, { label: string; variant: "default" | "secondary" | "destructive" | "outline" }> = {
      consolidated: { label: 'Terminé', variant: 'default' },
      collecting: { label: 'Collecte', variant: 'secondary' },
      ai_running: { label: 'Analyse IA', variant: 'secondary' },
      consolidating: { label: 'Consolidation', variant: 'secondary' },
      new: { label: 'Nouveau', variant: 'outline' },
      pending: { label: 'En attente', variant: 'outline' },
      error: { label: 'Erreur', variant: 'destructive' },
    };
    return statusConfig[status] || { label: status, variant: 'outline' as const };
  };

  const getAccountStatusBadge = (status: string) => {
    const statusConfig: Record<string, { label: string; variant: "default" | "secondary" | "destructive" }> = {
      active: { label: 'Actif', variant: 'default' },
      pending: { label: 'En attente', variant: 'secondary' },
      rejected: { label: 'Rejeté', variant: 'destructive' },
    };
    return statusConfig[status] || statusConfig.pending;
  };

  const freeAuditsRemaining = Math.max(0, 3 - (profile?.free_audits_used || 0));
  const canStartAudit = freeAuditsRemaining > 0 || profile?.account_status === 'active';

  if (loading) {

export const exportToPDF = async (title: string, elementId: string) => {
  try {
    // Dynamic imports pour réduire le bundle
    const [{ default: html2canvas }, { jsPDF }] = await Promise.all([
      import('html2canvas'),
      import('jspdf'),
    ]);

    const element = document.getElementById(elementId);
    if (!element) {
      throw new Error('Element not found');
    }

    const canvas = await html2canvas(element, {
      scale: 2,
      useCORS: true,
      logging: false,
    });

    const imgData = canvas.toDataURL('image/png');
    const pdf = new jsPDF({
      orientation: 'portrait',
      unit: 'mm',
      format: 'a4',
    });

    const imgWidth = 210;
    const pageHeight = 297;
    const imgHeight = (canvas.height * imgWidth) / canvas.width;
    let heightLeft = imgHeight;
    let position = 0;

    pdf.addImage(imgData, 'PNG', 0, position, imgWidth, imgHeight);
    heightLeft -= pageHeight;

    while (heightLeft >= 0) {
      position = heightLeft - imgHeight;
      pdf.addPage();
      pdf.addImage(imgData, 'PNG', 0, position, imgWidth, imgHeight);
      heightLeft -= pageHeight;
    }

    pdf.save(`${title.replace(/[^a-z0-9]/gi, '-').toLowerCase()}.pdf`);
  } catch (error) {
    console.error('Erreur lors de la génération du PDF:', error);
    throw error;
  }
};
